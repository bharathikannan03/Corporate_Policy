public function dataupload(Request $request)
    {
        $user = Auth::user();

        // Broker access control: brokers can only upload data for their assigned policies
        if ($this->isBrokerUser($user) &&
            ! $this->brokerCanAccessPolicy($user, (int) $request->policy_id)) {
            return response()->json(['message' => 'Unauthorized'], 403);
        }

        // Validation
        $rules = [
            'policy_data_type' => 'required|string',
            'ref_data_type_id' => 'required|integer',
            'remark' => 'nullable|string',
            'policy_data_upload' => 'nullable|array|min:1',
            'policy_data_upload.*' => 'required|file',
            'policy_id' => 'nullable|integer',
        ];

        // Custom validation for file types based on data type
        $messages = [
            'policy_data_upload.*.required' => 'File is required',
            'policy_data_upload.*.file' => 'The policy_data_upload field must be a file',
        ];

        $validator = Validator::make($request->all(), $rules, $messages);

        // Add custom validation for file extensions
        $validator->after(function ($validator) use ($request) {
            if ($request->hasFile('policy_data_upload')) {
                $ref_data_type_id = (int) $request->ref_data_type_id;
                $allowedExtensions = ($ref_data_type_id === 4) ? ['pdf', 'zip'] : ['xlsx', 'xls', 'csv'];

                foreach ($request->file('policy_data_upload') as $index => $file) {
                    $extension = strtolower($file->getClientOriginalExtension());
                    if (! in_array($extension, $allowedExtensions)) {
                        $validator->errors()->add(
                            "policy_data_upload.{$index}",
                            "The policy_data_upload.{$index} field must be a file of type: " . implode(', ', $allowedExtensions) . '.'
                        );
                    }
                }
            }
        });

        if ($validator->fails()) {
            return response()->json([
                'message' => $validator->errors()->all(),
            ], 400);
        }

        // Handle file upload
        $filepath = '';
        $fileName = '';
        
        if ($request->hasFile('policy_data_upload')) {
            $files = $request->file('policy_data_upload');
            
            // Store only the first file
            if (is_array($files) && count($files) > 0) {
                $file = $files[0];
                $storedPath = $file->store('data_upload', 's3');
                
                if ($storedPath === false) {
                    return response()->json([
                        'message' => 'File upload failed. Please check storage configuration.',
                    ], 500);
                }
                
                $filepath = $storedPath;
                $fileName = $file->getClientOriginalName();
            } elseif ($files instanceof \Illuminate\Http\UploadedFile) {
                // Single file case
                $storedPath = $files->store('data_upload', 's3');
                
                if ($storedPath === false) {
                    return response()->json([
                        'message' => 'File upload failed. Please check storage configuration.',
                    ], 500);
                }
                
                $filepath = $storedPath;
                $fileName = $files->getClientOriginalName();
            } else {
                return response()->json([
                    'message' => 'File not uploaded',
                ], 400);
            }
        } else {
            return response()->json([
                'message' => 'File not uploaded',
            ], 400);
        }

        // Store data in MasterPolicyDataUpload
        $policydataupload = MasterPolicyDataUpload::create([
            'data_type' => $request->policy_data_type,
            'remark' => $request->remark,
            'policy_data_upload' => $filepath ?? '',
            'original_file_name' => $fileName ?? '',
            'ref_policy_id' => $request->policy_id,
            'status' => 1,
            'is_dataupload' => 1,
            'created_by' => $user->id,
            'updated_by' => $user->id,
        ]);

        if (! $policydataupload) {
            return response()->json(['message' => 'Failed to save policy data upload'], 500);
        }

        // Check policy completion status
        $policyId = $request->policy_id;
        $sectionId = 4;

        if ($policyId) {
            $addpolicy = new MasterAddPolicy;
            $isCompleted = $addpolicy->getPolicyCompletion($policyId, $sectionId);

            // Update policy status
            $getpolicy = MasterAddPolicy::find($policyId);
            if ($getpolicy) {
                $getpolicy->update(['status' => $isCompleted]);
            }

            // Determine the import class based on ref_data_type_id
            $importClass = null;
            switch ($request->ref_data_type_id) {
                case 1:  // Inception Data
                    $importClass = new InceptionDataImport($request->policy_id, $policydataupload->id);
                    break;
                case 2:  // Endorsement Data
                    $importClass = new EndorsementDataImport($request->policy_id, $policydataupload->id);
                    break;
                case 3:  // Total Claim Report Data
                    $importClass = new TotalClaimReportImport($request->policy_id, $policydataupload->id);
                    break;

                case 9:  // Nominee Data
                    $importClass = new NomineeImport($request->policy_id, $policydataupload->id);
                    break;

                case 4: // Ecards Data (Strict: direct PDF or ZIP with only PDFs)
                    $files = $request->file('policy_data_upload') ?? [];
                    $invalidFiles = [];
                    $validFiles = [];
                    $seenCodes = [];

                    $validEmployeeCodes = DB::table('master_inception_data_uploads')
                        ->where('ref_policy_id', $request->policy_id)
                        ->select('employee_code')
                        ->union(
                            DB::table('master_endorsement_data_uploads')
                                ->where('ref_policy_id', $request->policy_id)
                                ->select('employee_code')
                        )
                        ->pluck('employee_code')
                        ->map(fn ($code) => strtoupper(trim($code)))
                        ->toArray();

                    $validEmployeeCodeLookup = array_flip($validEmployeeCodes);

                    $policyIdForEcards = (int) $request->policy_id;

                    $processPdf = function (string $pdfPath, string $originalName) use (&$validFiles, &$invalidFiles, &$seenCodes, $validEmployeeCodeLookup, $policyIdForEcards) {
                        $fileCode = strtoupper(trim(pathinfo($originalName, PATHINFO_FILENAME)));

                        if (! isset($validEmployeeCodeLookup[$fileCode])) {
                            $invalidFiles[] = $originalName.' (employee code not found)';

                            return;
                        }

                        // prevent duplicate employee_code in same request
                        if (isset($seenCodes[$fileCode])) {
                            $invalidFiles[] = $originalName.' (duplicate employee code in upload)';

                            return;
                        }
                        $seenCodes[$fileCode] = true;

                        $s3Path = 'ecard_uploads/'.$policyIdForEcards.'/'.$fileCode.'.pdf';

                        $stream = fopen($pdfPath, 'r');
                        if ($stream === false) {
                            $invalidFiles[] = $originalName.' (unable to read file)';

                            return;
                        }

                        try {
                            Storage::disk('s3')->put($s3Path, $stream, ['ContentType' => 'application/pdf']);
                        } finally {
                            fclose($stream);
                        }

                        $validFiles[] = [
                            'employee_code' => $fileCode,
                            'file_path' => $s3Path,
                        ];
                    };

                    $cleanupDir = function (string $dir): void {
                        if (! is_dir($dir)) {
                            return;
                        }
                        $it = new \RecursiveDirectoryIterator($dir, \FilesystemIterator::SKIP_DOTS);
                        $ri = new \RecursiveIteratorIterator($it, \RecursiveIteratorIterator::CHILD_FIRST);
                        foreach ($ri as $f) {
                            $f->isDir() ? @rmdir($f->getPathname()) : @unlink($f->getPathname());
                        }
                        @rmdir($dir);
                    };

                    foreach ($files as $uploadedFile) {
                        $ext = strtolower($uploadedFile->getClientOriginalExtension());

                        // Direct PDF
                        if ($ext === 'pdf') {
                            $processPdf($uploadedFile->getRealPath(), $uploadedFile->getClientOriginalName());

                            continue;
                        }

                        // ZIP (strict)
                        if ($ext === 'zip') {
                            $extractDir = storage_path('app/tmp/ecards_'.uniqid());

                            if (! is_dir($extractDir) && ! mkdir($extractDir, 0775, true) && ! is_dir($extractDir)) {
                                $invalidFiles[] = $uploadedFile->getClientOriginalName().' (cannot create temp folder)';

                                continue;
                            }

                            $zip = new \ZipArchive;
                            if ($zip->open($uploadedFile->getRealPath()) !== true) {
                                $invalidFiles[] = $uploadedFile->getClientOriginalName().' (invalid zip)';
                                $cleanupDir($extractDir);

                                continue;
                            }

                            $zipPdfCount = 0;
                            $zipHasNonPdf = false;

                            for ($i = 0; $i < $zip->numFiles; $i++) {
                                $entry = $zip->statIndex($i);
                                $entryName = $entry['name'] ?? '';

                                // folder entry
                                if (substr($entryName, -1) === '/') {
                                    continue;
                                }

                                $entryExt = strtolower(pathinfo($entryName, PATHINFO_EXTENSION));
                                if ($entryExt !== 'pdf') {
                                    $zipHasNonPdf = true;
                                    $invalidFiles[] = $uploadedFile->getClientOriginalName().' (contains non-pdf: '.$entryName.')';
                                    break;
                                }

                                $zipPdfCount++;
                            }

                            if ($zipHasNonPdf || $zipPdfCount === 0) {
                                if ($zipPdfCount === 0 && ! $zipHasNonPdf) {
                                    $invalidFiles[] = $uploadedFile->getClientOriginalName().' (zip has no pdf)';
                                }
                                $zip->close();
                                $cleanupDir($extractDir);

                                continue;
                            }

                            $zip->extractTo($extractDir);
                            $zip->close();

                            $iterator = new \RecursiveIteratorIterator(
                                new \RecursiveDirectoryIterator($extractDir, \FilesystemIterator::SKIP_DOTS)
                            );

                            foreach ($iterator as $item) {
                                if (! $item->isFile()) {
                                    continue;
                                }

                                if (strtolower($item->getExtension()) !== 'pdf') {
                                    $invalidFiles[] = $uploadedFile->getClientOriginalName().' (contains non-pdf after extract)';

                                    continue;
                                }

                                $processPdf($item->getPathname(), $item->getFilename());
                            }

                            $cleanupDir($extractDir);

                            continue;
                        }

                        $invalidFiles[] = $uploadedFile->getClientOriginalName().' (only pdf/zip allowed)';
                    }

                    if (! empty($invalidFiles)) {
                        $invalidFiles = array_values(array_unique($invalidFiles));

                        MasterEcardsDataUploadErrors::updateOrCreate(
                            [
                                'ref_policy_id' => $request->policy_id,
                            ],
                            [
                                'errors' => json_encode($invalidFiles),
                                'status' => false,
                            ]
                        );

                        return response()->json([
                            'status' => false,
                            'message' => 'Upload failed due to invalid files.',
                            'invalid_files' => $invalidFiles,
                        ], 422);
                    }

                    if (empty($validFiles)) {
                        return response()->json(['message' => 'No valid files found.'], 400);
                    }

                    foreach ($validFiles as $item) {
                        MasterEcardsDataUpload::updateOrCreate(
                            [
                                'employee_code' => $item['employee_code'],
                                'ref_policy_id' => $request->policy_id,
                            ],
                            [
                                'ecards_data_url' => $item['file_path'],
                                'created_at' => now(),
                            ]
                        );
                    }

                    return response()->json([
                        'status' => true,
                        'message' => 'E-card files uploaded successfully.',
                        'section_id' => $sectionId,
                    ], 200);

                    break;

                default:
                    return response()->json(['message' => 'Invalid document type'], 400);
            }

            // Import the Excel file
            Excel::import($importClass, $file);

            // Retrieve errors from the database
            // $failures = MasterDatauploadError::where('created_at', '>=', now()->subMinutes(5))->get();
            $failures = MasterDatauploadError::where('created_at', '>=', now()->subSeconds(10))->get();

            if ($failures->isNotEmpty()) {
                // Update status to 'Failed (2)' if errors exist
                $policydataupload->update(['is_dataupload' => 2]);

                return response()->json([
                    'message' => 'Policy Data Upload Failed',
                    'policy_data_upload' => $policydataupload,
                    'section_id' => $sectionId,
                ], 200);
            }

            return response()->json([
                'message' => 'Policy Data Uploaded successfully',
                'policy_data_upload' => $policydataupload,
                'section_id' => $sectionId,
            ], 200);
        }
    }
